 38 # Generate a sample dataset
 39 def make_realistic_regression(
 40     n_samples=3000,
 41     n_features=20,
 42     n_informative=15,
 43     noise=10.0,
 44     bias=10.0,
 45     corr_strength=0.15,
 46     nonlinear_strength=0.5,
 47     outlier_fraction=0.05,
 48     missing_fraction=0.0,
 49     categorical_features=6,
 50     random_state=None,
 51 ):
 52     """
 53     Generate a more realistic synthetic regression dataset.
 54
 55     Parameters
 56     ----------
 57     n_samples : int
 58         Number of samples.
 59     n_features : int
 60         Number of numerical features.
 61     n_informative : int
 62         Number of informative (truly predictive) features.
 63     noise : float
 64         Standard deviation of Gaussian noise added to the target.
 65     bias : float
 66         Constant bias term in target generation.
 67     corr_strength : float
 68         Std. dev. of correlated noise added to create redundancy.
 69     nonlinear_strength : float
 70         Strength of nonlinear feature transformations (0 disables).
 71     outlier_fraction : float
 72         Fraction of samples with outlier targets.
 73     missing_fraction : float
 74         Fraction of missing entries (NaN) in X.
 75     categorical_features : int
 76         Number of discrete/categorical columns to append.
 77     random_state : int or None
 78         Random seed for reproducibility.
 79     """
 80
 81     rng = np.random.default_rng(random_state)
 82
 83     # Step 1: Linear base dataset
 84     X, y = make_regression(
 85         n_samples=n_samples,
 86         n_features=n_features,
 87         n_informative=n_informative,
 88         noise=noise,
 89         bias=bias,
 90         random_state=random_state,
 91     )
 92
 93     # Step 2: Add nonlinear transformations
 94     if nonlinear_strength > 0:
 95         X_nl = X.copy()
 96         X_nl[:, 0] = np.exp(nonlinear_strength * 0.01 * X[:, 0])
 97         X_nl[:, 1] = np.sin(nonlinear_strength * X[:, 1])
 98         X_nl[:, 2] = (X[:, 2] ** 2) * nonlinear_strength
 99         X = np.hstack([X, X_nl])
100
101     # Step 3: Add correlated (redundant) features
102     if corr_strength > 0:
103         corr_features = X[:, :min(5, X.shape[1])] + rng.normal(
104             0, corr_strength, size=(n_samples, min(5, X.shape[1]))
105         )
106         X = np.hstack([X, corr_features])
107
108     # Step 4: Add outliers in target
109     if outlier_fraction > 0:
110         n_outliers = int(n_samples * outlier_fraction)
111         idx = rng.choice(n_samples, n_outliers, replace=False)
112         y[idx] += rng.normal(0, noise * 10, size=n_outliers)
113
114     # Step 5: Add missing values
115     if missing_fraction > 0:
116         missing_mask = rng.random(X.shape) < missing_fraction
117         X[missing_mask] = np.nan
118
119     # Step 6: Add categorical/discrete features
120     if categorical_features > 0:
121         cats = rng.integers(0, 3, size=(n_samples, categorical_features))
122         X = np.hstack([X, cats])
123
124     return X, y
125

127     random_state=0,
128 )

137 def add_missing_values(X_full, y_full, rng):
138     n_samples, n_features = X_full.shape
139
140     # Add missing values in 75% of the lines
141     missing_rate = 0.75
142     n_missing_samples = int(n_samples * missing_rate)
143
144     missing_samples = np.zeros(n_samples, dtype=bool)
145     missing_samples[:n_missing_samples] = True
146
147     rng.shuffle(missing_samples)
148     missing_features = rng.randint(0, n_features, n_missing_samples)
149     X_missing = X_full.copy()
150     X_missing[missing_samples, missing_features] = np.nan
151     y_missing = y_full.copy()
152
153     return X_missing, y_missing

177 def get_score(X, y, imputer=None):
178     regressor = RandomForestRegressor(random_state=0)
179     if imputer is not None:
180         estimator = make_pipeline(imputer, regressor)
181     else:
182         estimator = regressor
183     scores = cross_val_score(
184         estimator, X, y, scoring="neg_mean_squared_error", cv=N_SPLITS
185     )
186     return scores.mean(), scores.std()
